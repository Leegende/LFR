tank_designers_free = {
	if = {
		limit = {
			NOT = {
				has_country_flag = nation_picked_tank_designer
			}
		}
		ROOT = {
			add_ideas = free_tank_designer
			set_country_flag = nation_picked_tank_designer
			army_experience = 10
		}
	}
}
